package kr.gyul.yak;

import android.app.AlarmManager;
import android.app.PendingIntent;
import android.content.Context;
import android.content.Intent;
import android.content.SharedPreferences;
import android.os.Build;

import org.json.JSONArray;
import org.json.JSONObject;

import java.text.SimpleDateFormat;
import java.util.Calendar;
import java.util.Date;
import java.util.Locale;

public class Sched {
    static SharedPreferences p(Context c) { return c.getSharedPreferences("gyul", Context.MODE_PRIVATE); }

    static String day(long ms) { return new SimpleDateFormat("yyyyMMdd", Locale.KOREA).format(new Date(ms)); }

    static JSONObject cfg(Context c) {
        try { return new JSONObject(p(c).getString("cfg", "{}")); } catch (Exception e) { return new JSONObject(); }
    }

    static void scheduleAll(Context c) {
        try {
            JSONArray sl = cfg(c).optJSONArray("slots");
            if (sl == null) return;
            AlarmManager am = (AlarmManager) c.getSystemService(Context.ALARM_SERVICE);
            if (am == null) return;
            for (int i = 0; i < sl.length() && i < 10; i++) {
                JSONObject s = sl.getJSONObject(i);
                String id = s.optString("id");
                String t = s.optString("time", "");
                if (!t.matches("\\d{1,2}:\\d{2}")) continue;
                set(c, am, id, "dose", next(t, 0), 100 + i * 2);
                set(c, am, id, "miss", next(t, 30), 101 + i * 2);
            }
        } catch (Exception e) { }
    }

    static long next(String t, int addMin) {
        String[] a = t.split(":");
        Calendar k = Calendar.getInstance();
        k.set(Calendar.HOUR_OF_DAY, Integer.parseInt(a[0]));
        k.set(Calendar.MINUTE, Integer.parseInt(a[1]));
        k.set(Calendar.SECOND, 0);
        k.set(Calendar.MILLISECOND, 0);
        k.add(Calendar.MINUTE, addMin);
        if (k.getTimeInMillis() <= System.currentTimeMillis() + 5000) k.add(Calendar.DAY_OF_MONTH, 1);
        return k.getTimeInMillis();
    }

    static void set(Context c, AlarmManager am, String id, String kind, long at, int rc) {
        Intent i = new Intent(c, AlarmReceiver.class).putExtra("slot", id).putExtra("kind", kind);
        PendingIntent pi = PendingIntent.getBroadcast(c, rc, i,
                PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE);
        if (Build.VERSION.SDK_INT >= 31 && !am.canScheduleExactAlarms()) {
            am.setAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, at, pi);
        } else {
            am.setExactAndAllowWhileIdle(AlarmManager.RTC_WAKEUP, at, pi);
        }
    }
}

package kr.gyul.yak;

import android.app.Notification;
import android.app.NotificationChannel;
import android.app.NotificationManager;
import android.app.PendingIntent;
import android.content.BroadcastReceiver;
import android.content.Context;
import android.content.Intent;
import android.content.res.AssetFileDescriptor;
import android.media.AudioAttributes;
import android.media.MediaPlayer;

import org.json.JSONArray;
import org.json.JSONObject;

import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.nio.charset.StandardCharsets;

public class AlarmReceiver extends BroadcastReceiver {
    static final String CH = "dose2";

    @Override
    public void onReceive(Context c, Intent in) {
        final PendingResult pr = goAsync();
        final Context ctx = c.getApplicationContext();
        String slot = in.getStringExtra("slot");
        String kind = in.getStringExtra("kind");
        boolean async = false;
        Sched.p(ctx).edit().putString("last", kind + " " + slot + " " + new java.text.SimpleDateFormat("MM/dd HH:mm:ss", java.util.Locale.KOREA).format(new java.util.Date())).apply();
        try {
            JSONObject o = Sched.cfg(ctx);
            String who = o.optString("who", "gm");
            String topic = o.optString("topic", "");
            JSONObject s = find(o.optJSONArray("slots"), slot);
            String nm = s == null ? "약" : s.optString("nm", "약");
            String time = s == null ? "" : s.optString("time", "");
            String myName = o.optString("myName", "").trim();
            String name = "gm".equals(who) ? "할머니" : "gp".equals(who) ? "할아버지" : ("me".equals(who) && !myName.isEmpty()) ? myName + "님" : "";
            String vprefix = "me".equals(who) ? "c" : who;
            if ("dose".equals(kind)) {
                String title = name.isEmpty() ? nm + " 드실 시간이에요! 💊" : name + "~ " + nm + " 드실 시간이에요! 💊";
                notifyDose(ctx, title, "눌러서 귤손녀 앱을 열고 '약 먹었어요'를 눌러 주세요.", slot);
                async = play(ctx, vprefix + "_" + slot, pr);
            } else if ("miss".equals(kind)) {
                String d = Sched.day(System.currentTimeMillis() - 31 * 60 * 1000L);
                boolean taken = Sched.p(ctx).getBoolean("t_" + d + "_" + slot, false);
                if (!taken && !topic.isEmpty()) {
                    String subj = "gm".equals(who) ? "할머니께서" : "gp".equals(who) ? "할아버지께서" : name.isEmpty() ? "가족분이" : name + "이";
                    final JSONObject m = new JSONObject();
                    m.put("topic", topic);
                    m.put("title", "⚠️ " + subj + " " + nm + "을 아직 안 드셨어요");
                    m.put("message", time + " 약인데 30분이 지나도록 버튼을 안 누르셨어요. 전화 한 통 드려 보세요.");
                    m.put("priority", 5);
                    m.put("sequence_id", "m-" + d + "-" + slot);
                    m.put("tags", new JSONArray().put("warning"));
                    async = true;
                    new Thread(() -> { post(m.toString()); pr.finish(); }).start();
                }
            }
        } catch (Exception e) {
        } finally {
            Sched.scheduleAll(ctx);
            if (!async) pr.finish();
        }
    }

    static JSONObject find(JSONArray a, String id) {
        if (a == null) return null;
        for (int i = 0; i < a.length(); i++) {
            JSONObject s = a.optJSONObject(i);
            if (s != null && id != null && id.equals(s.optString("id"))) return s;
        }
        return null;
    }

    static void notifyDose(Context c, String title, String text, String slot) {
        NotificationManager nm = (NotificationManager) c.getSystemService(Context.NOTIFICATION_SERVICE);
        if (nm == null) return;
        NotificationChannel ch = new NotificationChannel(CH, "약 시간 알림", NotificationManager.IMPORTANCE_HIGH);
        ch.setSound(null, null);
        ch.enableVibration(true);
        ch.setVibrationPattern(new long[]{0, 600, 300, 600});
        nm.createNotificationChannel(ch);
        Intent open = new Intent(c, MainActivity.class).addFlags(Intent.FLAG_ACTIVITY_NEW_TASK | Intent.FLAG_ACTIVITY_CLEAR_TOP);
        PendingIntent pi = PendingIntent.getActivity(c, 7, open, PendingIntent.FLAG_UPDATE_CURRENT | PendingIntent.FLAG_IMMUTABLE);
        Notification n = new Notification.Builder(c, CH)
                .setSmallIcon(android.R.drawable.ic_lock_idle_alarm)
                .setContentTitle(title)
                .setContentText(text)
                .setStyle(new Notification.BigTextStyle().bigText(text))
                .setCategory(Notification.CATEGORY_REMINDER)
                .setContentIntent(pi)
                .setAutoCancel(true)
                .build();
        try { nm.notify(("n" + slot).hashCode(), n); } catch (Exception e) { }
    }

    static boolean play(Context c, String key, PendingResult pr) {
        int id = c.getResources().getIdentifier(key, "raw", c.getPackageName());
        if (id == 0) id = c.getResources().getIdentifier("c_" + key.substring(key.indexOf('_') + 1), "raw", c.getPackageName());
        if (id == 0) return false;
        try {
            MediaPlayer mp = new MediaPlayer();
            mp.setAudioAttributes(new AudioAttributes.Builder()
                    .setUsage(AudioAttributes.USAGE_ALARM)
                    .setContentType(AudioAttributes.CONTENT_TYPE_SPEECH).build());
            AssetFileDescriptor afd = c.getResources().openRawResourceFd(id);
            mp.setDataSource(afd.getFileDescriptor(), afd.getStartOffset(), afd.getLength());
            afd.close();
            mp.prepare();
            mp.setOnCompletionListener(m -> { m.release(); pr.finish(); });
            mp.setOnErrorListener((m, w, x) -> { m.release(); pr.finish(); return true; });
            mp.start();
            return true;
        } catch (Exception e) {
            return false;
        }
    }

    static void post(String body) {
        try {
            HttpURLConnection h = (HttpURLConnection) new URL("https://ntfy.sh/").openConnection();
            h.setRequestMethod("POST");
            h.setDoOutput(true);
            h.setConnectTimeout(10000);
            h.setReadTimeout(10000);
            h.setRequestProperty("Content-Type", "application/json");
            OutputStream os = h.getOutputStream();
            os.write(body.getBytes(StandardCharsets.UTF_8));
            os.close();
            h.getResponseCode();
            h.disconnect();
        } catch (Exception e) { }
    }
}
